# Function
make_loo_plot <- function(dat) {
  xmin <- min(dat$LCI, na.rm = TRUE)
  xmax <- max(dat$UCI, na.rm = TRUE)  
  x_lower <- floor(xmin / 0.20) * 0.20
  x_upper <- ceiling(xmax / 0.20) * 0.20
  x_breaks <- seq(from = x_lower,to   = x_upper,by   = 0.20)
  ggplot(dat,aes(x = OR, y = SNP) 
         ) + 
    geom_errorbarh(
      aes(xmin = LCI,xmax = UCI
      ),
      height = 0.15,
      linewidth = 0.55
    ) +
    geom_point(size = 2.1
    ) +
    geom_vline(
      xintercept = 1,
      colour = "grey60",
      linetype = "solid",
      linewidth = 0.6
    ) +
    scale_x_continuous(
      breaks = x_breaks,
      labels = function(x) sprintf("%.1f", x),
      expand = expansion(mult = c(0.02, 0.02))
    ) + 
    coord_cartesian(
      xlim = c(x_lower, x_upper),
      clip = "off"
    ) +
    labs(
      x = "OR (95% CI)",
      y = NULL
    ) +
    theme_classic(
      base_size = 12,
      base_family = "A"
    ) +
    theme(
      text = element_text(
        family = "A",
        colour = "black"
      ),
      plot.title = element_blank(),
      axis.text.y = element_text(
        family = "A",
        size = 9,
        colour = "black"
      ),
      axis.text.x = element_text(
        family = "A",
        size = 10,
        colour = "black"
      ),
      
      axis.title.x = element_text(
        family = "A",
        size = 11,
        colour = "black"
      ),
      axis.title.y = element_blank(),
      plot.margin = margin(t = 5.5,r = 5.5,b = 5.5,l = 5.5
      )
    )
}
p_dep <- make_loo_plot(loo_dep_or)
